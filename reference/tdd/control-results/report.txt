python3 - <<'PY'
import os
from pathlib import Path
import shutil
import subprocess
import tempfile

source = Path.cwd()
temp_base = os.path.realpath(tempfile.gettempdir())
with tempfile.TemporaryDirectory(dir=temp_base) as directory:
    copy = Path(directory)
    shutil.copytree(source / 'template/process', copy / 'template/process')
    shutil.copy2(source / 'Makefile', copy / 'Makefile')
    patch = source / 'reference/tdd/control-results/fix.patch'
    env = dict(os.environ, TMPDIR=temp_base)
    for phase, flags, expected in [('RED', ['--reverse'], 2), ('GREEN', [], 0)]:
        subprocess.run(['git', 'apply', *flags, str(patch)], cwd=copy, check=True)
        result = subprocess.run(['make', 'test'], cwd=copy, env=env,
                                capture_output=True, text=True)
        print(phase, 'exit:', result.returncode, flush=True)
        print(result.stdout + result.stderr, flush=True)
        assert result.returncode == expected
PY
